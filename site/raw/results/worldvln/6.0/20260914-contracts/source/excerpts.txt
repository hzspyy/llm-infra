  121  bool Work::wait(std::chrono::milliseconds timeout) {
  122    std::unique_lock<std::mutex> lock(mutex_);
  123    if (timeout == kNoTimeout) {
  124      // This waits without a timeout.
  125      cv_.wait(lock, [&] { return completed_; });
  126    } else {
  127      // Waits for the user-provided timeout.
  128      cv_.wait_for(lock, timeout, [&] { return completed_; });
  129      if (!completed_) {
  130        // Throw exception if the wait operation timed out and the work was not
  131        // completed.
  132        TORCH_CHECK(false, "Operation timed out!");
  133      }
  134    }
  135    if (exception_) {
  136      std::rethrow_exception(exception_);
  137    }
  138    synchronize();
  139    // Always return true, because abort API is not implemented.
  140    return true;
  141  }

  721  void ProcessGroupGloo::runLoop(int workerIndex) {
  722    c10::setThreadName("pt_gloo_runloop");
  723    std::unique_lock<std::mutex> lock(workMutex_);
  724  
  725    while (!stop_) {
  726      if (workQueue_.empty()) {
  727        workProduceCV_.wait(lock);
  728        continue;
  729      }
  730  
  731      auto work = std::move(workQueue_.front());
  732      workQueue_.pop_front();
  733      workInProgress_[workerIndex] = work;
  734      lock.unlock();
  735  
  736      // Notify after releasing the lock so that the waiter
  737      // does not immediately block.
  738      workConsumeCV_.notify_one();
  739  
  740      AsyncWork::execute(work);
  741      // TODO: Need to find a way to calculate the difference of duration of two
  742      // c10d::Event
  743      pgStatus_->lastCompletedSeq = static_cast<int64_t>(work->seq_);
  744      pgStatus_->lastCompletedWorkName = opTypeToString(work->opType_);
  745      // TODO: We need to have numel of tensors for gloo as well.
  746      pgStatus_->lastCompletedNumelIn = 0;
  747      pgStatus_->lastCompletedNumelOut = 0;
  748      FlightRecorder<c10::Event>::get()->retire_id(
  749          work->trace_id_, work->trace_reset_epoch_, false);
  750      lock.lock();
  751      workInProgress_[workerIndex].reset();
