#!/usr/bin/env python3
"""L1.2 补测 · 分析 ncu 采集的 FP8 cuBLAS 指令计数器

解读 ncu 输出，计算实际的 tensor core 指令发射率，
并与 tensor_core.cu 测得的 mma.sync.m16n8k32 基准对比。

目标：确定 cuBLAS FP8 是否使用了 mxf8f6f4.block_scale 或其他更宽的指令。
"""
import csv
import json
from pathlib import Path


def parse_ncu_csv(csv_path):
    """解析 ncu CSV，提取所有指标"""
    with open(csv_path) as f:
        reader = csv.DictReader(f)
        rows = list(reader)

    metrics = {}
    for row in rows:
        kernel = row.get("Kernel Name", "")
        metric = row.get("Metric Name", "")
        value = row.get("Metric Value", "")
        unit = row.get("Metric Unit", "")

        if not kernel or not metric:
            continue

        if kernel not in metrics:
            metrics[kernel] = {}

        # 移除千位分隔符
        if value:
            value = value.replace(",", "")

        metrics[kernel][metric] = {"value": value, "unit": unit}

    return metrics


def analyze_fp8_gemm(metrics, sm_count=170):
    """分析 FP8 GEMM 的指令计数器"""

    print("=== L1.2 FP8 cuBLAS 指令计数器分析 ===\n")

    if not metrics:
        print("ERROR: No metrics found in CSV")
        return

    # 找到主 GEMM kernel（通常包含 gemm 或 xmma 字样）
    gemm_kernels = [k for k in metrics.keys() if "gemm" in k.lower() or "xmma" in k.lower()]

    if not gemm_kernels:
        print("WARNING: No GEMM kernel found, analyzing all kernels:")
        gemm_kernels = list(metrics.keys())

    for kernel_name in gemm_kernels:
        print(f"Kernel: {kernel_name}")
        print("=" * 80)

        m = metrics[kernel_name]

        # 提取关键指标
        duration_ns = float(m.get("gpu__time_duration.sum", {}).get("value", 0))
        tensor_inst = float(m.get("sm__inst_executed_pipe_tensor.sum", {}).get("value", 0))
        hmma_inst = float(m.get("sm__inst_executed_pipe_tensor_op_hmma.sum", {}).get("value", 0))
        dmma_inst = float(m.get("sm__inst_executed_pipe_tensor_op_dmma.sum", {}).get("value", 0))
        sm_util = m.get("sm__throughput.avg.pct_of_peak_sustained_elapsed", {}).get("value", "N/A")

        print(f"\n基础指标：")
        print(f"  Kernel duration:        {duration_ns/1e6:>10.3f} ms")
        print(f"  SM utilization:         {sm_util:>10s} %")
        print(f"\nTensor core 指令计数：")
        print(f"  总 tensor 指令:         {tensor_inst:>10.0f}")
        print(f"  hmma 指令 (warp级):     {hmma_inst:>10.0f}")
        print(f"  dmma 指令:              {dmma_inst:>10.0f}")

        # 计算每条指令的 FLOP
        # 已知：mma.sync.m16n8k32 = 8192 FLOP/inst
        #      mma.sync.m16n8k64 = 16384 FLOP/inst (如果存在)
        #      mxf8f6f4.block_scale 可能有不同的 FLOP/inst

        # 假设测试的是 8192x8192x8192 GEMM
        total_flop = 2 * (8192 ** 3)

        if tensor_inst > 0:
            flop_per_inst = total_flop / tensor_inst
            print(f"\n推导：")
            print(f"  总 FLOP (8192³×2):      {total_flop/1e12:>10.3f} TFLOP")
            print(f"  FLOP per tensor inst:   {flop_per_inst:>10.0f}")

            # 已知的指令 FLOP
            known_instructions = {
                8192: "mma.sync.m16n8k32.e4m3 (标准)",
                16384: "mma.sync.m16n8k64.e4m3 (更宽) 或 mxf8f6f4",
                4096: "mma.sync.m16n8k16.bf16 (降级?)",
            }

            match = known_instructions.get(int(flop_per_inst), "未知指令")
            print(f"  推测使用的指令:        {match}")

            if duration_ns > 0:
                # 假设某个时钟频率（需要从 verify_gemm_ceiling.py 的结果获取）
                # 这里先用理论值
                assumed_ghz = 2.1  # cuBLAS 在功耗墙下的典型频率

                cycles = duration_ns * assumed_ghz  # ns * GHz = cycles
                inst_per_cycle = tensor_inst / cycles
                inst_per_cycle_per_sm = inst_per_cycle / sm_count

                print(f"\n发射率分析（假设 {assumed_ghz} GHz）：")
                print(f"  总周期:                 {cycles:>10.0f}")
                print(f"  Inst per cycle per SM:  {inst_per_cycle_per_sm:>10.4f}")

                # 与基准对比
                baseline_m16n8k32 = 1022  # FLOP/clk/SM from tensor_core.cu
                actual_flop_per_clk_per_sm = (total_flop / 1e12) / (duration_ns / 1e9) / assumed_ghz / sm_count * 1e12

                print(f"\n与基准对比：")
                print(f"  实测 FLOP/clk/SM:       {actual_flop_per_clk_per_sm:>10.0f}")
                print(f"  基准 (m16n8k32):        {baseline_m16n8k32:>10.0f}")
                print(f"  超出比例:               {actual_flop_per_clk_per_sm/baseline_m16n8k32:>10.2f}x")

                if actual_flop_per_clk_per_sm > baseline_m16n8k32 * 1.1:
                    print(f"\n⚠️  实测超出基准 {(actual_flop_per_clk_per_sm/baseline_m16n8k32 - 1)*100:.1f}%")
                    print(f"    可能原因：")
                    print(f"      1. cuBLAS 使用了更宽的 FP8 指令 (m16n8k64 或 mxf8f6f4)")
                    print(f"      2. 假设的时钟频率不准确")
                    print(f"      3. 使用了 f16 累加而非 f32（理论上可以 2× 吞吐）")

        print("\n" + "=" * 80 + "\n")


def main():
    import sys

    if len(sys.argv) < 2:
        csv_path = Path("results/spark/ncu/12_fp8_cublas.csv")
    else:
        csv_path = Path(sys.argv[1])

    if not csv_path.exists():
        print(f"ERROR: {csv_path} not found")
        print(f"Usage: {sys.argv[0]} [path/to/12_fp8_cublas.csv]")
        sys.exit(1)

    print(f"Reading: {csv_path}\n")

    metrics = parse_ncu_csv(csv_path)
    analyze_fp8_gemm(metrics)

    # 输出 JSON 供后续分析
    json_path = csv_path.with_suffix(".analysis.json")
    with open(json_path, "w") as f:
        json.dump(metrics, f, indent=2)
    print(f"详细数据已保存到: {json_path}")


if __name__ == "__main__":
    main()
