#!/usr/bin/env python3
"""L1.2 补测 · 用 Nsight Compute 采集 cuBLAS FP8 的 tensor core 指令计数器

问题：verify_gemm_ceiling.py 测得 cuBLAS FP8 达到 1501 FLOP/clk/SM，
      超过 tensor_core.cu 的纯 mma.sync.m16n8k32 上限 1022 FLOP/clk/SM（47% 超出）

假设：cuBLASLt 可能使用了更宽的 FP8 指令（如 mxf8f6f4.block_scale）

验证：用 ncu 采集实际执行的 tensor core 指令计数器：
      - sm__inst_executed_pipe_tensor* 系列
      - 具体的 mma 指令分类计数
      - 累加精度（f16 vs f32）

环境：spark (GB10, sm_121) 有 ncu 权限
输出：results/spark/ncu/12_fp8_cublas.csv 和 .ncu-rep
"""
import subprocess
import sys
import json
from pathlib import Path

import torch


def check_env():
    """检查是否在 spark 且有 ncu 权限"""
    try:
        # 检查是否有 ncu 命令
        result = subprocess.run(["which", "ncu"], capture_output=True, text=True, check=False)
        if result.returncode != 0:
            print("ERROR: ncu not found. This script must run on spark with ncu available.")
            sys.exit(1)

        # 检查是否是 sm_121
        p = torch.cuda.get_device_properties(0)
        if p.major != 12 or p.minor != 1:
            print(f"WARNING: Expected sm_121, got sm_{p.major}{p.minor}. Results may differ.")

        print(f"✓ Running on {p.name}, sm_{p.major}{p.minor}, {p.multi_processor_count} SMs")
        return True
    except Exception as e:
        print(f"ERROR: Environment check failed: {e}")
        sys.exit(1)


def run_gemm_with_ncu(size=8192, output_dir=".", prefix="12_fp8_cublas"):
    """运行 FP8 GEMM 并用 ncu 采集指令计数器"""

    print(f"\n=== Preparing {size}x{size}x{size} FP8 GEMM ===")

    # 准备输入（与 verify_gemm_ceiling.py 相同）
    af = torch.randn(size, size, device="cuda") / 8
    bf = torch.randn(size, size, device="cuda") / 8
    a8 = af.to(torch.float8_e4m3fn)
    b8 = bf.to(torch.float8_e4m3fn).t().contiguous().t()
    sc = torch.tensor(1.0, device="cuda")

    # 预热，让 kernel 编译完成
    print("Warming up...")
    for _ in range(5):
        _ = torch._scaled_mm(a8, b8, scale_a=sc, scale_b=sc, out_dtype=torch.bfloat16)
    torch.cuda.synchronize()

    # 正确性检查
    ref = (af.to(torch.bfloat16) @ bf.to(torch.bfloat16)).float()
    got = torch._scaled_mm(a8, b8, scale_a=sc, scale_b=sc, out_dtype=torch.bfloat16).float()
    rel_err = ((got - ref).abs().mean() / ref.abs().mean()).item()
    print(f"✓ Correctness: relative error {rel_err:.4f} (bf16 mean {ref.abs().mean():.4f}, "
          f"fp8 mean {got.abs().mean():.4f})")

    # 写一个简单的启动脚本供 ncu 调用
    launcher = Path(output_dir) / f"{prefix}_launch.py"
    launcher.write_text(f"""#!/usr/bin/env python3
import torch
a8 = torch.randn({size}, {size}, device="cuda", dtype=torch.float8_e4m3fn)
b8 = torch.randn({size}, {size}, device="cuda", dtype=torch.float8_e4m3fn).t().contiguous().t()
sc = torch.tensor(1.0, device="cuda")
# 只跑一次，让 ncu 捕获
result = torch._scaled_mm(a8, b8, scale_a=sc, scale_b=sc, out_dtype=torch.bfloat16)
torch.cuda.synchronize()
""")
    launcher.chmod(0o755)

    # 关键指标：tensor core 指令计数器
    metrics = [
        "sm__inst_executed_pipe_tensor.sum",  # 总的 tensor core 指令数
        "sm__inst_executed_pipe_tensor_op_hmma.sum",  # hmma 指令（warp级）
        "sm__inst_executed_pipe_tensor_op_dmma.sum",  # dmma 指令（如果有）
        "smsp__inst_executed_pipe_tensor.sum",  # 每个子分区的 tensor 指令
        "gpu__time_duration.sum",  # 总时间
        "sm__throughput.avg.pct_of_peak_sustained_elapsed",  # SM 利用率
    ]

    csv_out = Path(output_dir) / f"{prefix}.csv"
    rep_out = Path(output_dir) / f"{prefix}.ncu-rep"

    ncu_cmd = [
        "ncu",
        "--metrics", ",".join(metrics),
        "--csv",
        "--log-file", str(csv_out),
        "--export", str(rep_out),
        "--target-processes", "all",
        sys.executable, str(launcher)
    ]

    print(f"\n=== Running ncu (this may take 1-2 minutes) ===")
    print(f"Command: {' '.join(ncu_cmd)}")

    result = subprocess.run(ncu_cmd, capture_output=True, text=True, check=False)

    if result.returncode != 0:
        print(f"ERROR: ncu failed with exit code {result.returncode}")
        print(f"STDERR:\n{result.stderr}")
        return False

    print(f"✓ ncu completed")
    print(f"  CSV: {csv_out}")
    print(f"  Report: {rep_out}")

    # 清理临时启动脚本
    launcher.unlink()

    return True


def parse_and_summarize(csv_path):
    """解析 ncu CSV 并提取关键数据"""
    import csv

    print(f"\n=== Parsing {csv_path} ===")

    with open(csv_path) as f:
        reader = csv.DictReader(f)
        rows = list(reader)

    if not rows:
        print("WARNING: No data in CSV")
        return

    # 提取关键指标
    summary = {}
    for row in rows:
        metric = row.get("Metric Name", "")
        value = row.get("Metric Value", "")
        if metric and value:
            summary[metric] = value
            print(f"  {metric}: {value}")

    # 计算 FLOP/clk/SM（如果有足够信息）
    if "gpu__time_duration.sum" in summary:
        ns = float(summary["gpu__time_duration.sum"].replace(",", ""))
        print(f"\n  Kernel duration: {ns/1e6:.3f} ms")


def main():
    check_env()

    # 确定输出目录
    # 如果在 spark 的学习目录，输出到 work/out/ncu/
    # 否则输出到当前目录
    output_dir = Path("/home/smc/learn/work/out/ncu")
    if not output_dir.exists():
        output_dir = Path(".")
        print(f"WARNING: {output_dir} not found, using current directory")
    else:
        output_dir.mkdir(parents=True, exist_ok=True)

    print(f"Output directory: {output_dir}")

    # 运行 ncu 采集（与 verify_gemm_ceiling.py 一致使用 8192）
    success = run_gemm_with_ncu(size=8192, output_dir=output_dir, prefix="12_fp8_cublas")

    if success:
        csv_path = output_dir / "12_fp8_cublas.csv"
        if csv_path.exists():
            parse_and_summarize(csv_path)
        print("\n✓ 实验完成。请将 results/spark/ncu/12_fp8_cublas.* 文件拉回本地分析。")
    else:
        print("\n✗ 实验失败")
        sys.exit(1)


if __name__ == "__main__":
    main()
