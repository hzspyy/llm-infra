#!/bin/bash
# 本地检查：验证 L1.2 FP8 实验文件的完整性
#
# 运行：bash labs/L1/check_fp8_files.sh

set -euo pipefail

echo "=== L1.2 FP8 实验文件完整性检查 ==="
echo ""

LABS_DIR="labs/L1"
FILES=(
    "ncu_fp8_gemm.py"
    "run_ncu_fp8.sh"
    "analyze_fp8_ncu.py"
    "tensor_core_fp8_variants.cu"
    "README_fp8_investigation.md"
)

echo "检查文件存在性："
all_exist=true
for f in "${FILES[@]}"; do
    if [[ -f "$LABS_DIR/$f" ]]; then
        size=$(du -h "$LABS_DIR/$f" | cut -f1)
        echo "  ✓ $f ($size)"
    else
        echo "  ✗ $f (缺失)"
        all_exist=false
    fi
done
echo ""

if [[ "$all_exist" != "true" ]]; then
    echo "ERROR: 部分文件缺失"
    exit 1
fi

echo "检查 Python 语法："
PYTHON="/Volumes/data/venvs/llm-infra/bin/python"
for f in ncu_fp8_gemm.py analyze_fp8_ncu.py; do
    if $PYTHON -m py_compile "$LABS_DIR/$f" 2>/dev/null; then
        echo "  ✓ $f"
    else
        echo "  ✗ $f (语法错误)"
        exit 1
    fi
done
echo ""

echo "检查 shell 脚本："
if [[ -x "$LABS_DIR/run_ncu_fp8.sh" ]]; then
    echo "  ✓ run_ncu_fp8.sh (可执行)"
else
    echo "  ✗ run_ncu_fp8.sh (不可执行)"
    exit 1
fi
echo ""

echo "检查关键配置一致性："
# 检查矩阵大小
size_in_ncu=$(grep -o "size=[0-9]*" "$LABS_DIR/ncu_fp8_gemm.py" | head -1 | cut -d= -f2)
size_in_analyze=$(grep -o "[0-9]* \*\* 3" "$LABS_DIR/analyze_fp8_ncu.py" | head -1 | awk '{print $1}')

if [[ "$size_in_ncu" == "$size_in_analyze" ]]; then
    echo "  ✓ 矩阵大小一致: ${size_in_ncu}x${size_in_ncu}x${size_in_ncu}"
else
    echo "  ✗ 矩阵大小不一致: ncu=$size_in_ncu, analyze=$size_in_analyze"
    exit 1
fi
echo ""

echo "=== 所有检查通过 ==="
echo ""
echo "下一步："
echo "  1. 同步到 spark:"
echo "     rsync -av $LABS_DIR/ncu_fp8_gemm.py $LABS_DIR/run_ncu_fp8.sh $LABS_DIR/tensor_core_fp8_variants.cu spark:/home/smc/learn/work/labs/L1/"
echo ""
echo "  2. SSH 到 spark 并运行:"
echo "     ssh spark"
echo "     cd /home/smc/learn"
echo "     source env.sh"
echo "     bash work/labs/L1/run_ncu_fp8.sh"
echo ""
echo "  3. 编译并运行 CUDA 变体测试:"
echo "     cd /home/smc/learn/work/labs/L1"
echo "     nvcc -O3 -std=c++17 -arch=sm_121a -o tensor_core_fp8_variants tensor_core_fp8_variants.cu"
echo "     ./tensor_core_fp8_variants > ../../out/12_fp8_variants.txt"
echo ""
echo "  4. 拉回结果:"
echo "     rsync -av --ignore-existing spark:/home/smc/learn/work/out/ncu/12_fp8_cublas.* results/spark/ncu/"
echo "     rsync -av --ignore-existing spark:/home/smc/learn/work/out/12_fp8_variants.txt results/spark/"
