#!/bin/bash
# L1.2 补测 · spark 上运行 FP8 cuBLAS 的 ncu 采集
#
# 使用方法：
#   1. 本地同步脚本到 spark：
#      rsync -av labs/L1/ncu_fp8_gemm.py spark:/home/smc/learn/work/labs/L1/
#   2. SSH 到 spark 并运行：
#      ssh spark
#      cd /home/smc/learn
#      source env.sh
#      bash work/labs/L1/run_ncu_fp8.sh
#   3. 拉回结果：
#      rsync -av --ignore-existing spark:/home/smc/learn/work/out/ncu/12_fp8_cublas.* results/spark/ncu/

set -euo pipefail

# 确保在 spark 学习目录
if [[ ! -f "/home/smc/learn/env.sh" ]]; then
    echo "ERROR: 必须在 spark 的学习目录运行此脚本"
    exit 1
fi

# 加载环境
source /home/smc/learn/env.sh

# 检查 Python 环境（spark 使用系统 python3）
PYTHON="python3"
if ! command -v "$PYTHON" &>/dev/null; then
    echo "ERROR: python3 not found in PATH"
    exit 1
fi

echo "=== L1.2 FP8 cuBLAS ncu 采集 ==="
echo "Python: $PYTHON"
echo "CUDA_HOME: ${CUDA_HOME:-not set}"
echo "HF_HOME: ${HF_HOME:-not set}"
echo ""

# 创建输出目录
mkdir -p /home/smc/learn/work/out/ncu

# 检查 GPU 状态
echo "=== GPU 状态 ==="
nvidia-smi --query-gpu=index,name,memory.used,memory.total,utilization.gpu --format=csv
echo ""

# 运行实验
cd /home/smc/learn/work/labs/L1
$PYTHON ncu_fp8_gemm.py

# 检查输出文件
echo ""
echo "=== 生成的文件 ==="
ls -lh /home/smc/learn/work/out/ncu/12_fp8_cublas.* 2>/dev/null || echo "没有找到输出文件"

echo ""
echo "=== 完成 ==="
echo "请拉回结果到本地："
echo "  rsync -av --ignore-existing spark:/home/smc/learn/work/out/ncu/12_fp8_cublas.* results/spark/ncu/"
