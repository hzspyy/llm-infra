#!/usr/bin/env python3
"""Custom JSON commit protocol with sequential rank records and failure injection.

This is not PyTorch DCP. It demonstrates hashes, staged publication and a latest
pointer in a single process. Real DCP behavior is checked separately by
checkpoint_runtime_contract.py. No multi-rank durability guarantee is inferred.
"""

import hashlib
import json
import os
import shutil
import sys
from dataclasses import asdict, dataclass
from pathlib import Path
from typing import Any, Dict, List, Optional, Tuple


@dataclass
class CheckpointMetadata:
    step: int
    world_size: int
    ranks_committed: List[int]
    shards_manifest: Dict[str, str]  # filename -> sha256
    model_config: Dict[str, Any]
    optimizer_type: str
    global_norm: float
    data_cursor_offset: int
    is_finalized: bool


class AtomicCheckpointer:
    """模拟支持原子提交与故障注入的分布式存储器"""

    def __init__(self, base_dir: Path):
        self.base_dir = base_dir
        self.base_dir.mkdir(parents=True, exist_ok=True)
        self.latest_link_file = self.base_dir / "latest_checkpoint.json"

    def save_checkpoint(
        self,
        step: int,
        world_size: int,
        rank_data: Dict[int, Dict[str, Any]],
        fault_mode: Optional[str] = None,
    ) -> Tuple[bool, str]:
        """
        执行分布式保存流程，并在指定阶段注入故障
        fault_mode 可选:
          - "FAULT_PRE_STAGING"
          - "FAULT_PARTIAL_SHARD"
          - "FAULT_PRE_MANIFEST"
          - "FAULT_PRE_LATEST"
          - None (正常)
        """
        step_dir = self.base_dir / f"checkpoint_step_{step}"
        staging_dir = self.base_dir / f".staging_step_{step}"

        if staging_dir.exists():
            shutil.rmtree(staging_dir)
        staging_dir.mkdir(parents=True, exist_ok=True)

        if fault_mode == "FAULT_PRE_STAGING":
            # 故障 1：准备阶段崩溃，无任何数据写入
            return False, "SimulatedCrash: Process terminated before staging shards"

        # 1. 各 Rank 独立写入分片数据
        shards_manifest = {}
        for rank in range(world_size):
            shard_path = staging_dir / f"rank_{rank}_shard.bin"
            data = rank_data.get(rank, {"rank": rank, "dummy_weights": [0.1 * rank]})

            if fault_mode == "FAULT_PARTIAL_SHARD" and rank == 1:
                # 故障 2：Rank 1 写入中途崩溃，文件截断
                with open(shard_path, "wb") as f:
                    f.write(b"CORRUPTED_INCOMPLETE_TRUNCATED_BYTES")
                return False, "SimulatedCrash: I/O error or kernel panic while writing rank 1 shard"

            serialized = json.dumps(data, sort_keys=True).encode("utf-8")
            with open(shard_path, "wb") as f:
                f.write(serialized)
            sha256 = hashlib.sha256(serialized).hexdigest()
            shards_manifest[shard_path.name] = sha256

        # 2. 阶段二：生成并提交 .metadata 清单
        if fault_mode == "FAULT_PRE_MANIFEST":
            # 故障 3：分片均已落盘，但在元数据清单持久化前崩溃
            return False, "SimulatedCrash: Node coordinator killed before finalizing .metadata manifest"

        metadata = CheckpointMetadata(
            step=step,
            world_size=world_size,
            ranks_committed=list(range(world_size)),
            shards_manifest=shards_manifest,
            model_config={"layers": 36, "hidden_size": 3072},
            optimizer_type="AdamW",
            global_norm=1.024,
            data_cursor_offset=step * 512,
            is_finalized=True,
        )

        metadata_path = staging_dir / ".metadata"
        with open(metadata_path, "w", encoding="utf-8") as f:
            json.dump(asdict(metadata), f, indent=2)

        # 原子重命名完成阶段提交 (Atomic directory rename)
        if step_dir.exists():
            shutil.rmtree(step_dir)
        staging_dir.rename(step_dir)

        # 3. 阶段三：更新全局指针
        if fault_mode == "FAULT_PRE_LATEST":
            # 故障 4：检查点已完整落盘，但在更新全局 latest 指针前崩溃
            return False, "SimulatedCrash: Process died before updating latest_checkpoint.json"

        with open(self.latest_link_file, "w", encoding="utf-8") as f:
            json.dump(
                {
                    "latest_step": step,
                    "checkpoint_dir": str(step_dir.name),
                    "committed_shards": len(shards_manifest),
                },
                f,
                indent=2,
            )

        return True, "Checkpoint successfully committed and published"

    def resolve_latest_valid_checkpoint(self) -> Optional[Dict[str, Any]]:
        """
        稳健的检查点恢复解析器：
        1. 优先读取 latest_checkpoint 指针
        2. 校验目标目录的合法性 (必须存在 .metadata 且各个分片 SHA256 完整对齐)
        3. 若指针损坏或目标不合法，降级扫描所有 checkpoint_step_* 目录，寻找合法的最大 step
        4. 忽略任何遗留的 .staging_* 目录
        """
        candidate_step = None
        if self.latest_link_file.exists():
            try:
                with open(self.latest_link_file, "r", encoding="utf-8") as f:
                    latest_info = json.load(f)
                    candidate_step = latest_info.get("latest_step")
            except Exception:
                candidate_step = None

        if candidate_step is not None:
            cand_dir = self.base_dir / f"checkpoint_step_{candidate_step}"
            is_valid, meta = self._validate_checkpoint_dir(cand_dir)
            if is_valid:
                return {"step": candidate_step, "dir": str(cand_dir), "metadata": meta, "fallback": False}

        # 触发回退降级扫描 (Fallback Scan)
        all_dirs = [d for d in self.base_dir.iterdir() if d.is_dir() and d.name.startswith("checkpoint_step_")]
        valid_checkpoints = []
        for d in all_dirs:
            try:
                step_num = int(d.name.replace("checkpoint_step_", ""))
                is_valid, meta = self._validate_checkpoint_dir(d)
                if is_valid:
                    valid_checkpoints.append((step_num, d, meta))
            except ValueError:
                continue

        if not valid_checkpoints:
            return None

        # 选出步数最大的合法检查点
        valid_checkpoints.sort(key=lambda x: x[0], reverse=True)
        best_step, best_dir, best_meta = valid_checkpoints[0]
        return {
            "step": best_step,
            "dir": str(best_dir),
            "metadata": best_meta,
            "fallback": True,
        }

    def _validate_checkpoint_dir(self, step_dir: Path) -> Tuple[bool, Optional[Dict[str, Any]]]:
        if not step_dir.exists() or not step_dir.is_dir():
            return False, None
        meta_file = step_dir / ".metadata"
        if not meta_file.exists():
            return False, None

        try:
            with open(meta_file, "r", encoding="utf-8") as f:
                meta = json.load(f)
            if not meta.get("is_finalized", False):
                return False, None

            # 校验各 rank 分片是否存在且未受损
            shards_manifest = meta.get("shards_manifest", {})
            for shard_name, expected_sha in shards_manifest.items():
                shard_path = step_dir / shard_name
                if not shard_path.exists():
                    return False, None
                with open(shard_path, "rb") as sf:
                    actual_sha = hashlib.sha256(sf.read()).hexdigest()
                if actual_sha != expected_sha:
                    return False, None

            return True, meta
        except Exception:
            return False, None


def main():
    import tempfile
    from _evidence import new_output, write_result
    output_dir = new_output('Custom JSON checkpoint commit protocol')
    test_dir = Path(tempfile.mkdtemp(prefix='json-checkpoint-', dir=os.environ['TMPDIR']))

    checkpointer = AtomicCheckpointer(test_dir)
    world_size = 4

    print("=== 单进程自定义 JSON 提交协议故障模拟（不调用 DCP） ===")

    # 1. 正常保存 Step 1000 作为安全基线
    rank_data_1000 = {
        r: {"rank": r, "step": 1000, "tensors": f"tensor_weights_rank_{r}"} for r in range(world_size)
    }
    success, msg = checkpointer.save_checkpoint(
        step=1000, world_size=world_size, rank_data=rank_data_1000, fault_mode=None
    )
    print(f"Step 1000 保存: success={success} ({msg})")

    resolved_1000 = checkpointer.resolve_latest_valid_checkpoint()
    print(f"  -> 当前最新合法 Checkpoint: Step {resolved_1000['step']}")

    # 2. 依次注入四种故障并检验回退安全性
    injection_results = {}
    fault_scenarios = [
        ("FAULT_PRE_STAGING", 2000, "序列化前崩溃"),
        ("FAULT_PARTIAL_SHARD", 2000, "分片写途中崩溃(截断损坏)"),
        ("FAULT_PRE_MANIFEST", 2000, "分片完成但元数据提交前崩溃"),
        ("FAULT_PRE_LATEST", 2000, "元数据提交完成但指针未更新"),
    ]

    for fault_mode, target_step, desc in fault_scenarios:
        print(f"\n[注入测试] 场景: {fault_mode} ({desc}) @ Step {target_step}")
        rank_data = {
            r: {"rank": r, "step": target_step, "tensors": f"tensor_weights_step_{target_step}_rank_{r}"}
            for r in range(world_size)
        }
        success, msg = checkpointer.save_checkpoint(
            step=target_step, world_size=world_size, rank_data=rank_data, fault_mode=fault_mode
        )
        print(f"  保存执行结果: success={success} | 拦截消息: {msg}")

        # 解析最新检查点
        resolved = checkpointer.resolve_latest_valid_checkpoint()
        if resolved is None:
            resolved_info = {"status": "NONE"}
        else:
            resolved_info = {
                "active_step": resolved["step"],
                "fallback_triggered": resolved["fallback"],
            }
        print(f"  恢复定位结果: Step {resolved_info.get('active_step')}, 回退降级: {resolved_info.get('fallback_triggered')}")

        injection_results[fault_mode] = {
            "description": desc,
            "target_step": target_step,
            "save_success": success,
            "error_msg": msg,
            "resolved_step": resolved_info.get("active_step"),
            "is_corrupt_chosen": (resolved_info.get("active_step") == target_step and not success and fault_mode != "FAULT_PRE_LATEST"),
        }

    # 3. 正常保存 Step 3000，验证指针恢复与正常演进
    print("\n[恢复测试] 正常保存 Step 3000")
    rank_data_3000 = {
        r: {"rank": r, "step": 3000, "tensors": f"tensor_weights_step_3000_rank_{r}"} for r in range(world_size)
    }
    success, msg = checkpointer.save_checkpoint(
        step=3000, world_size=world_size, rank_data=rank_data_3000, fault_mode=None
    )
    resolved_3000 = checkpointer.resolve_latest_valid_checkpoint()
    print(f"  -> 当前最新合法 Checkpoint: Step {resolved_3000['step']}, 回退状态: {resolved_3000['fallback']}")

    assert resolved_3000['step'] == 3000
    assert all(not row['is_corrupt_chosen'] for row in injection_results.values())
    write_result(output_dir, 'checkpoint_failure_injection_report.json', {
        'test_suite': 'Custom_JSON_Atomic_Commit_Failure_Injection',
        'logical_shards': world_size, 'real_worker_processes': 0,
        'dcp_called': False, 'baseline_step': 1000,
        'scenarios': injection_results, 'post_recovery_step': resolved_3000['step'],
    }, {'faults': [row[0] for row in fault_scenarios],
        'execution': 'single-process custom JSON protocol'}, [__file__])

    # 清理临时测试目录
    shutil.rmtree(test_dir)


if __name__ == "__main__":
    main()
