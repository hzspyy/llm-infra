#!/usr/bin/env python3
"""Two-rank unequal-denominator update versus a global FP64 reference.

Run with torchrun. NCCL supports both DDP and FSDP2; Gloo checks DDP math.
Each selected mode performs one update. No throughput/quality claim is made.
"""
import argparse
import copy
import json
import os
from pathlib import Path
import torch
import torch.distributed as dist
from torch import nn


def materialize(value):
    return value.full_tensor() if hasattr(value,'full_tensor') else value


def network():
    return nn.Sequential(nn.Linear(4,8,bias=False),nn.Tanh(),nn.Linear(8,3,bias=False))


def one_update(mode,device,rank,world):
    torch.manual_seed(0)
    model=network().to(device=device,dtype=torch.float64)
    reference=copy.deepcopy(model)
    x=torch.randn(5,4,device=device,dtype=torch.float64)
    y=torch.randn(5,3,device=device,dtype=torch.float64)
    mask=torch.ones_like(y)
    mask[0,0]=0
    split=slice(0,2) if rank==0 else slice(2,5)
    ref_opt=torch.optim.AdamW(reference.parameters(),lr=.001,weight_decay=.01,foreach=False,fused=False)
    ref_loss=((reference(x)-y).square()*mask).sum()/mask.sum()
    ref_loss.backward()
    ref_grads={n:p.grad.detach().clone() for n,p in reference.named_parameters()}
    ref_norm=torch.nn.utils.clip_grad_norm_(reference.parameters(),.5)
    ref_opt.step()
    if mode=='ddp':
        wrapped=nn.parallel.DistributedDataParallel(model,device_ids=[device.index] if device.type=='cuda' else None)
    else:
        from torch.distributed.fsdp import fully_shard
        for layer in model:
            if any(True for _ in layer.parameters()):fully_shard(layer)
        fully_shard(model)
        wrapped=model
    opt=torch.optim.AdamW(model.parameters(),lr=.001,weight_decay=.01,foreach=False,fused=False)
    local_count=mask[split].sum()
    global_count=local_count.clone()
    returned=dist.all_reduce(global_count,op=dist.ReduceOp.SUM)
    assert returned is None
    local_sum=((wrapped(x[split])-y[split]).square()*mask[split]).sum()
    loss=local_sum*world/global_count
    loss.backward()
    errors={}
    for name,p in model.named_parameters():
        grad=materialize(p.grad)
        errors[name]=float((grad-ref_grads[name]).abs().max())
        torch.testing.assert_close(grad,ref_grads[name],rtol=0,atol=1e-12)
    norm=torch.nn.utils.clip_grad_norm_(model.parameters(),.5)
    norm=float(materialize(norm))
    assert abs(norm-float(ref_norm))<1e-12
    opt.step()
    parameter_errors={}
    state_errors={}
    ref_params=dict(reference.named_parameters())
    for name,p in model.named_parameters():
        full=materialize(p.detach())
        rp=ref_params[name]
        parameter_errors[name]=float((full-rp.detach()).abs().max())
        torch.testing.assert_close(full,rp.detach(),rtol=0,atol=1e-12)
        state_errors[name]={}
        for key in ('exp_avg','exp_avg_sq'):
            actual=materialize(opt.state[p][key])
            expected=ref_opt.state[rp][key]
            state_errors[name][key]=float((actual-expected).abs().max())
            torch.testing.assert_close(actual,expected,rtol=0,atol=1e-12)
    # A deliberately wrong mean-of-rank-means control, computed without collectives.
    torch.manual_seed(0)
    bad=network().to(device=device,dtype=torch.float64)
    bad_loss=(((bad(x[:2])-y[:2]).square()*mask[:2]).sum()/mask[:2].sum()+
              ((bad(x[2:])-y[2:]).square()*mask[2:]).sum()/mask[2:].sum())/2
    bad_loss.backward()
    wrong_error=max(float((p.grad-ref_grads[n]).abs().max()) for n,p in bad.named_parameters())
    assert wrong_error>1e-6
    result={'mode':mode,'rank':rank,'local_count':float(local_count),'global_count':float(global_count),
            'all_reduce_return':returned,'gradient_errors':errors,'parameter_errors':parameter_errors,
            'optimizer_state_errors':state_errors,'global_grad_norm':norm,
            'mean_of_local_means_error':wrong_error,'parameter_layouts':[
                {'name':n,'global_shape':list(p.shape),'local_shape':list(p.to_local().shape) if hasattr(p,'to_local') else list(p.shape),
                 'placements':str(getattr(p,'placements','replicated'))} for n,p in model.named_parameters()]}
    del wrapped,model,reference,opt,ref_opt,bad
    if device.type=='cuda':torch.cuda.synchronize();torch.cuda.empty_cache()
    return result


def skip_contract(device,rank):
    p=nn.Parameter(torch.tensor(1.,device=device))
    opt=torch.optim.SGD([p],lr=.1)
    p.grad=torch.tensor(float('inf') if rank==1 else 2.,device=device)
    flag=torch.isfinite(p.grad).to(torch.int32)
    dist.all_reduce(flag,op=dist.ReduceOp.MIN)
    before=p.detach().clone()
    if flag.item():opt.step()
    assert torch.equal(p.detach(),before)
    count=torch.tensor(0.,device=device)
    dist.all_reduce(count)
    return {'rank':rank,'global_finite':int(flag),'step_skipped':True,'parameter':float(p.detach()),
            'all_zero_target_count':float(count),'zero_target_action':'skip on every rank; no division'}


def main():
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--outdir',required=True,type=Path)
    parser.add_argument('--backend',choices=['gloo','nccl'],default='gloo')
    parser.add_argument('--modes',nargs='+',choices=['ddp','fsdp'],default=['ddp'])
    args=parser.parse_args()
    rank=int(os.environ['RANK']);local_rank=int(os.environ['LOCAL_RANK'])
    device=torch.device('cuda',local_rank) if args.backend=='nccl' else torch.device('cpu')
    if device.type=='cuda':torch.cuda.set_device(device)
    torch.set_num_threads(1)
    dist.init_process_group(args.backend)
    try:
        world=dist.get_world_size()
        assert world==2
        if rank==0:args.outdir.mkdir(parents=True,exist_ok=False)
        dist.barrier()
        rows=[one_update(mode,device,rank,world) for mode in args.modes]
        result={'torch':torch.__version__,'torch_git':torch.version.git_version,'backend':args.backend,
                'rank':rank,'world_size':world,'updates':rows,'skip':skip_contract(device,rank)}
        (args.outdir/f'rank-{rank}.json').write_text(json.dumps(result,indent=2)+'\n')
        dist.barrier()
        if rank==0:
            from _evidence import write_result
            gathered=[json.loads((args.outdir/f'rank-{i}.json').read_text()) for i in range(world)]
            write_result(args.outdir,'update_contract.json',{'ranks':gathered},
                         {'dtype':'float64','seed':0,'global_batch_shape':[5,4], 'valid_counts':[5,9],
                          'updates_per_rank':len(args.modes),'atol':1e-12,'rtol':0,
                          'performance_or_quality_measured':False},[__file__])
    finally:
        dist.destroy_process_group()


if __name__=='__main__':
    main()
