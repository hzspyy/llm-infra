#!/bin/bash
# Run from the project root after loading the environment in ENVIRONMENTS.md.
set -euo pipefail
: "${RUN_DIR:?Set RUN_DIR to a new experiment directory on the learning disk}"
PYTHON_BIN=${PYTHON_BIN:-python}
mkdir -p "$RUN_DIR"
test ! -e "$RUN_DIR/training-diagnostics.log"
if command -v nvidia-smi >/dev/null 2>&1; then
    nvidia-smi --query-compute-apps=pid,process_name,used_memory --format=csv
fi
set +e
"$PYTHON_BIN" labs/L7/training_loop_systems.py --experiment mixed_precision \
    --model HuggingFaceTB/SmolLM2-360M --outdir "$RUN_DIR/diagnostics" \
    2>&1 | tee "$RUN_DIR/training-diagnostics.log"
run_status=${PIPESTATUS[0]}
set -e
if command -v nvidia-smi >/dev/null 2>&1; then
    nvidia-smi --query-compute-apps=pid,process_name,used_memory --format=csv
fi
exit "$run_status"
