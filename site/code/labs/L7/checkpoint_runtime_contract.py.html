#!/usr/bin/env python3
"""One-process real DCP sync/async save contract and a missing-metadata failure.

Only small tensors are saved, in a caller-specified TMPDIR on the external disk.
This is separate from the custom JSON commit protocol in checkpoint_failure_injection.
"""
import copy
import hashlib
import json
import os
from pathlib import Path
import tempfile
import warnings
import torch
import torch.distributed.checkpoint as dcp


def run_dcp():
    torch.manual_seed(0)
    model=torch.nn.Linear(3,2,bias=False).double()
    optimizer=torch.optim.AdamW(model.parameters(),lr=.01,weight_decay=0.)
    x=torch.randn(2,3,dtype=torch.float64)
    model(x).square().mean().backward()
    optimizer.step()
    weight=model.weight
    payload={'weight':weight.detach().clone(), 'm':optimizer.state[weight]['exp_avg'].clone(),
             'v':optimizer.state[weight]['exp_avg_sq'].clone(), 'step':optimizer.state[weight]['step'].clone()}
    saved={k:v.clone() for k,v in payload.items()}
    with tempfile.TemporaryDirectory(prefix='dcp-contract-',dir=os.environ['TMPDIR']) as temp:
        root=Path(temp)
        with warnings.catch_warnings(record=True) as recorded:
            metadata=dcp.save(payload,checkpoint_id=root/'sync')
            future=dcp.async_save(payload,checkpoint_id=root/'async')
            if hasattr(future,'staging_completion'):
                future.staging_completion.result()
                upload=future.upload_completion
                staging='explicit staging_completion future'
            else:
                upload=future
                staging='default stager completes CPU staging before returning the upload Future'
            for value in payload.values():value.add_(100)
            upload.result()
            loaded={k:torch.empty_like(v) for k,v in saved.items()}
            dcp.load(loaded,checkpoint_id=root/'async')
            for key in saved:torch.testing.assert_close(loaded[key],saved[key],atol=0,rtol=0)
            files=[{'name':p.name,'bytes':p.stat().st_size,'sha256':hashlib.sha256(p.read_bytes()).hexdigest()}
                   for p in sorted((root/'async').iterdir())]
            # Damage only this test's newly created checkpoint, after preserving its file list.
            (root/'async'/'.metadata').unlink()
            try:dcp.load(loaded,checkpoint_id=root/'async')
            except BaseException as error:
                if isinstance(error,(KeyboardInterrupt,SystemExit)):raise
                failure={'type':type(error).__name__,'message':str(error)}
            else:raise AssertionError('DCP accepted a directory without metadata')
    return {'torch':torch.__version__,'torch_git':torch.version.git_version,
            'sync_return_type':type(metadata).__name__,'async_return_type':type(future).__name__,
            'staging_contract':staging,'snapshot_unchanged_after_source_mutation':True,
            'restored_tensors':{k:v.tolist() for k,v in saved.items()},'files':files,
            'missing_metadata_failure':failure,'warnings':[str(w.message) for w in recorded],
            'scope':'real single-process CPU DCP; no cross-rank or crash-durability guarantee inferred'}


def adam_reset_counterexample():
    p=torch.nn.Parameter(torch.tensor(1.,dtype=torch.float64))
    opt=torch.optim.Adam([p],lr=.1)
    p.grad=torch.ones_like(p);opt.step();opt.zero_grad(set_to_none=True)
    q=torch.nn.Parameter(p.detach().clone())
    fresh=torch.optim.Adam([q],lr=.1)
    p.grad=torch.ones_like(p);q.grad=torch.ones_like(q)
    opt.step();fresh.step()
    torch.testing.assert_close(p,q,rtol=0,atol=1e-12)
    return {'gradient_sequence':[1.,1.],'continued_parameter':p.item(),'reset_parameter':q.item(),
            'max_error':float((p-q).detach().abs()),
            'interpretation':'Resetting Adam changes state but does not necessarily enlarge the next update or create a loss spike.'}


def main():
    from _evidence import new_output,write_result
    out=new_output('Real DCP staging and update-state counterexamples')
    result={'dcp':run_dcp(),'adam_reset':adam_reset_counterexample()}
    write_result(out,'checkpoint_runtime.json',result,
                 {'device':'CPU','dtype':'float64','seed':0,'nn_updates':1,'dcp_world_size':1,
                  'fault':'remove metadata from a newly created temporary checkpoint',
                  'adam_counterexample_updates':2,'GPU_execution':False},[__file__])


if __name__=='__main__':
    main()
